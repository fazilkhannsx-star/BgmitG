function parseConfigString(raw) {
  let cleaned = raw.trim();
  if (cleaned.includes("=")) {
    cleaned = cleaned.substring(cleaned.indexOf("=") + 1).trim();
  }
  if (cleaned.endsWith(";")) cleaned = cleaned.slice(0, -1).trim();
  cleaned = cleaned.replace(/([{,]\s*)([a-zA-Z0-9_]+)\s*:/g, '$1"$2":').replace(/'/g, '"');
  return JSON.parse(cleaned);
}

function connectFirebase(raw, pushInitial) {
  try {
    const cfg = parseConfigString(raw);
    if (!firebase.apps.length) {
      firebase.initializeApp(cfg);
    }
    dbRef = firebase.database().ref("indra_esports_live");
    dbRef.on("value", snap => {
      const val = snap.val();
      if (val) {
        state = {
          ownerPin: val.ownerPin || state.ownerPin || DEFAULT_PIN,
          users: val.users || [],
          matches: val.matches || [],
          bookings: val.bookings || [],
          tickets: val.tickets || []
        };
        localStorage.setItem("indra_esports_state_v1", JSON.stringify(state));
        renderAll();
      } else if (pushInitial) {
        dbRef.set(state);
      }
      isCloudLive = true;
      document.getElementById("syncDot").classList.add("live");
      document.getElementById("syncText").textContent = "Live Cloud Synced";
    });
  } catch (e) {
    showToast("Invalid Firebase config format. Check JSON keys.");
  }
}

    title, map, mode, type, fee, prize, totalSlots, time,
    roomId: "", roomPass: ""
  });
  document.getElementById("mTitle").value = "";
  saveState();
  showToast("New Tournament Published Live!");
}

function verifyBooking(bookingId) {
  const b = state.bookings.find(x => x.id === bookingId);
  if (!b) return;
  b.status = "VERIFIED";
  if (!b.admitCode) {
    b.admitCode = String(Math.floor(1000 + Math.random() * 9000));
  }
  saveState();
  showToast(`Verified! 4-Digit Admit Code ${b.admitCode} issued to ${b.bgmiIgn}`);
}

function rejectBooking(bookingId) {
  const b = state.bookings.find(x => x.id === bookingId);
  if (!b) return;
  b.status = "REJECTED";
  saveState();
  showToast("Booking marked as Rejected.");
}

function saveRoomCredentials(matchId) {
  const m = state.matches.find(x => x.id === matchId);
  if (!m) return;
  m.roomId = document.getElementById("rid_" + matchId).value.trim();
  m.roomPass = document.getElementById("rpass_" + matchId).value.trim();
  saveState();
  showToast("Room ID & Password updated for verified players!");
}

function deleteMatch(matchId) {
  state.matches = state.matches.filter(m => m.id !== matchId);
  saveState();
  showToast("Tournament removed.");
}

function approveTicketWA(ticketId, approve) {
  const t = state.tickets.find(x => x.id === ticketId);
  if (!t) return;
  t.waApproved = approve;
  const rep = document.getElementById("trep_" + ticketId);
  if (rep) t.ownerReply = rep.value.trim();
  saveState();
  showToast(approve ? "WhatsApp permission granted to player!" : "Permission revoked.");
}
